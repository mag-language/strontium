1use crate::machine::{Executor, Strontium, StrontiumError};
2use crate::machine::register::RegisterValue;
3use super::super::CalculationMethod;
4use crate::Instruction;
5
6/// Perform a calculation two registers and write the result to a third.
7#[derive(Debug, Clone, PartialEq)]
8pub struct CalculateExecutor;
9
10impl Executor for CalculateExecutor {
11    fn execute(&self, machine: &mut Strontium) -> Result<bool, StrontiumError> {
12        if machine.debug {
13            println!("CALCULATE instruction");
14        }
15        let instruction = machine.parse_instruction()?;
16
17        if let Instruction::Calculate {
18            method,
19            operand1,
20            operand2,
21            destination,
22        } = instruction
23        {
24            let op1 = machine.registers.get(&operand1);
25            let op2 = machine.registers.get(&operand2);
26
27            if let (Some(op1), Some(op2)) = (op1, op2) {
28                let result = match method {
29                    CalculationMethod::ADD => op1.clone() + op2.clone(),
30                    CalculationMethod::SUBTRACT => op1.clone() - op2.clone(),
31                    CalculationMethod::MULTIPLY => op1.clone() * op2.clone(),
32                    CalculationMethod::DIVIDE => op1.clone() / op2.clone(),
33                    CalculationMethod::MODULO => op1.clone() % op2.clone(),
34                    CalculationMethod::POWER => match (op1.clone(), op2.clone()) {
35                        (RegisterValue::Int64(a), RegisterValue::Int64(b)) => RegisterValue::Int64(a.pow(b as u32)),
36                        (RegisterValue::Float64(a), RegisterValue::Float64(b)) => RegisterValue::Float64(a.powf(b)),
37                        _ => panic!("Incompatible types for power"),
38                    },
39                    CalculationMethod::SQRT => match op1.clone() {
40                        RegisterValue::Int64(a) => RegisterValue::Float64((a as f64).sqrt()),
41                        RegisterValue::Float64(a) => RegisterValue::Float64(a.sqrt()),
42                        RegisterValue::Float32(a) => RegisterValue::Float32(a.sqrt()),
43                        _ => panic!("Incompatible type for sqrt"),
44                    },
45                    CalculationMethod::CONCAT => match (op1.clone(), op2.clone()) {
46                        (RegisterValue::String(a), RegisterValue::String(b)) => {
47                            RegisterValue::String(a + &b)
48                        }
49                        _ => panic!("~ requires two String operands"),
50                    },
51                };
52
53                machine.registers.set(&destination, result);
54            }
55        }
56        Ok(true)
57    }
58}
